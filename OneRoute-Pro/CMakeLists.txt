cmake_minimum_required(VERSION 3.16)
project(ChennaiOneRoute VERSION 5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(transit_core src/engine.cpp)
target_include_directories(transit_core PUBLIC include)
if(MSVC)
 target_compile_options(transit_core PRIVATE /W4)
else()
 target_compile_options(transit_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(transit_engine src/main.cpp)
target_link_libraries(transit_engine PRIVATE transit_core)
enable_testing()
add_executable(transit_tests tests/test_engine.cpp)
target_link_libraries(transit_tests PRIVATE transit_core)
add_test(NAME engine COMMAND transit_tests)
add_executable(yen_tests tests/test_yen.cpp)
target_link_libraries(yen_tests PRIVATE transit_core)
add_test(NAME yen_oracle COMMAND yen_tests)
